## CODE STANDARDS (VTT 3.5)

```csharp
public class CollisionAbility : MonoBehaviour, IAbility
{
    [HideInInspector] public List<Collider> Collisions;
    public List<MonoBehaviour> CollisionActions;
    public List<IAbility> CollisionActionsAbilities = new();

    private void Start()
    {
        foreach (var action in CollisionActions)
        {
            if (action is IAbility ability)
                CollisionActionsAbilities.Add(ability);
            else
                Debug.LogError($"{action.name} must derive from IAbility");
        }
    }

    public void Execute()
    {
        foreach (var action in CollisionActionsAbilities)
            action.Execute();
    }
}
```

```csharp
public interface IAbilityTarget : IAbility
{
    List<GameObject> ActionTouches { get; set; }
}
```
