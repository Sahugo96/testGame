## Суть урока

**OnScreenStick** + апгрейд **Event System** на **Input System UI Input Module**. Разделение **`UserInputSystem`** (чтение `InputAction`, запись в ECS) и **`CharacterMoveSystem`** (применение движения). Hybrid bridge: MB `UserInputData` → struct **`InputData`** + tag **`MoveData`** через **`IConvertGameObjectToEntity`**. Движение по **Transform** (demo); скорость из Inspector → bake в **`MoveData`**.

## CODE STANDARDS (VTT 2.4)

```csharp
public class UserInputData : MonoBehaviour, IConvertGameObjectToEntity
{
    public float Speed = 10f;

    public void Convert(Entity entity, EntityManager dstManager, GameObjectConversionSystem conversionSystem)
    {
        dstManager.AddComponentData(entity, new InputData());
        dstManager.AddComponentData(entity, new MoveData { Speed = Speed / 100f });
    }
}

public struct InputData : IComponentData
{
    public float2 Move;
}

public struct MoveData : IComponentData
{
    public float Speed;
}
```

```csharp
public class UserInputSystem : ComponentSystemBase
{
    private EntityQuery _inputQuery;
    private InputAction _moveAction;
    private float2 _moveInput;

    protected override void OnCreate()
    {
        _inputQuery = GetEntityQuery(ComponentType.ReadWrite<InputData>());
    }

    protected override void OnStartRunning()
    {
        _moveAction = new InputAction("move");
        _moveAction.AddBinding("<Gamepad>/rightStick");
        _moveAction.AddCompositeBinding("2DVector")
            .With("Up", "<Keyboard>/w")
            .With("Down", "<Keyboard>/s")
            .With("Left", "<Keyboard>/a")
            .With("Right", "<Keyboard>/d");
        _moveAction.performed += ctx => _moveInput = ctx.ReadValue<Vector2>();
        _moveAction.canceled += ctx => _moveInput = float2.zero;
        _moveAction.Enable();
    }

    protected override void OnStopRunning() => _moveAction.Disable();

    protected override void OnUpdate()
    {
        Entities.With(_inputQuery).ForEach((ref InputData inputData) =>
        {
            inputData.Move = _moveInput;
        }).Run();
    }
}
```

```csharp
public class CharacterMoveSystem : ComponentSystemBase
{
    private EntityQuery _moveQuery;

    protected override void OnCreate()
    {
        _moveQuery = GetEntityQuery(
            ComponentType.ReadWrite<InputData>(),
            ComponentType.ReadOnly<MoveData>());
    }

    protected override void OnUpdate()
    {
        Entities.With(_moveQuery).ForEach(
            (Transform transform, in InputData input, in MoveData move) =>
            {
                var pos = transform.position;
                pos += new Vector3(input.Move.x, 0f, input.Move.y) * move.Speed;
                transform.position = pos;
            }).Run();
    }
}
```
